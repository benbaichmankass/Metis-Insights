from ml.experiments.splitters import split_live_holdout, split_time_aware_holdout
rows=[]
# backtest/cusum synthetic rows spanning Jan..Jun, live trades in Mar
for d in range(1,7):
    rows.append({"created_at":f"2026-0{d}-15T00:00:00Z","is_live_trade":False,"event_source":"backtest","f":d,"won":d%2})
rows.append({"created_at":"2026-03-10T00:00:00Z","is_live_trade":True,"event_source":"live","f":9.9,"won":1})
# paper-mirror twin of the same trade (same entry bar -> same features/outcome)
rows.append({"created_at":"2026-03-10T00:00:00Z","is_live_trade":False,"event_source":"live_paper","f":9.9,"won":1})
tr,ev=split_live_holdout(rows,{"live_flag_column":"is_live_trade","time_column":"created_at"})
print("eval max ts:",max(r["created_at"] for r in ev))
print("train rows AFTER last eval row:",[r["created_at"] for r in tr if r["created_at"]>max(x["created_at"] for x in ev)])
print("eval rows with identical-feature twin in train:",[(r["event_source"],r["f"]) for r in ev if any(t["f"]==r["f"] and t["won"]==r["won"] for t in tr)])
# paper-cohort manifest config (setup-candidates-metalabel-paper-v1)
tr2,ev2=split_live_holdout(rows,{"live_flag_column":"event_source","live_flag_true_value":"live_paper","time_column":"created_at"})
print("paper-eval: eval=",[r["event_source"] for r in ev2]," twin in train:",[t["event_source"] for t in tr2 if t["f"]==9.9])
