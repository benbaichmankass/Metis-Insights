import sys, json, math, os, tempfile, sqlite3, random
sys.path.insert(0, "/home/user/Metis-Insights")
from datetime import datetime, timedelta, timezone
from ml.datasets.families.exit_candidates import ExitCandidatesBuilder
from ml.datasets.families.setup_candidates import _load_live_trades
random.seed(1); d=tempfile.mkdtemp(); px=100.0; rows=[]
t0=datetime(2025,1,1,tzinfo=timezone.utc)
for i in range(200):
    o=px; c=o*math.exp(random.gauss(0,0.01)); rows.append({"ts":(t0+timedelta(hours=i)).isoformat(),"symbol":"BTCUSDT","timeframe":"1h","source":"t","open":o,"high":max(o,c)*1.002,"low":min(o,c)*0.998,"close":c,"volume":1}); px=c
open(os.path.join(d,"data.jsonl"),"w").write("\n".join(json.dumps(r) for r in rows))
db=os.path.join(d,"j.db"); con=sqlite3.connect(db)
con.execute("CREATE TABLE trades (id INTEGER PRIMARY KEY, timestamp TEXT, closed_at TEXT, symbol TEXT, direction TEXT, entry_price REAL, stop_loss REAL, position_size REAL, pnl REAL, pnl_percent REAL, status TEXT, is_backtest INT, is_demo INT, account_class TEXT, notes TEXT)")
# ONE paper-mirror row (account_class='paper', is_demo=0) with FABRICATED pnl provenance
con.execute("INSERT INTO trades VALUES (1,?,?,?,?,?,?,?,?,?,?,?,?,?,?)",((t0+timedelta(hours=50,minutes=3)).isoformat(),(t0+timedelta(hours=56)).isoformat(),"BTCUSDT","long",rows[50]["open"],rows[50]["open"]*0.98,1,3.0,1.0,"closed",0,0,"paper",json.dumps({"pnl_source":"synthesized"})))
con.commit(); con.close()
print("setup_candidates real loader keeps:", len(_load_live_trades(db,"BTCUSDT")))
out=[r for r in ExitCandidatesBuilder().iter_rows(market_raw_path=d, live_trades_db=db, include_synthetic=False)]
print("exit_candidates rows emitted from the paper/fabricated trade:", len(out), "is_live_trade values:", sorted({r["is_live_trade"] for r in out}), "event_source:", sorted({r["event_source"] for r in out}))
